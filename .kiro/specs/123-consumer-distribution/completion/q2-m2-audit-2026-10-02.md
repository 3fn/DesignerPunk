**The script, verbatim, so a reader can re-run it.** Usage: `python3 m2_audit.py <repo-root> <rev>`. It reads with `git show <rev>:<path>` and writes nothing.

```python
#!/usr/bin/env python3
"""Independent M2 (criteria parity) audit — Stacy, 2026-10-02. Does NOT import or call the
checker (scripts/completion-claims/*). Usage: python3 m2_audit.py <repo-root> <rev>
Reads tasks.md and task-<N>-completion.md via `git show <rev>:<path>`."""
import re, subprocess, sys, collections
root, rev = sys.argv[1], sys.argv[2]
SPEC = '.kiro/specs/123-consumer-distribution'
def show(p): return subprocess.run(['git', '-C', root, 'show', f'{rev}:{p}'], capture_output=True, text=True, check=True).stdout
def norm(s):  # the guide's rules (i)-(iii), re-implemented here; rule (iv): nothing else
    s = s.replace('\\|', '|')
    s = re.sub(r'<br\s*/?>', ' ', s, flags=re.I)
    s = re.sub('[​‌‍﻿]', '', s)
    return re.sub(r'\s+', ' ', s).strip()
BULLET = re.compile(r'^\s*[-*] (?!\[[ xX]\])(.*)$')
CHECKBOX = re.compile(r'^\s*[-*] \[[ xX]\] ')
LABEL = re.compile(r'^\s*\*\*[A-Z][^*]*:\*\*')
def criteria(tasks, n):
    lines = tasks.split('\n')
    start = next(i for i, l in enumerate(lines) if re.match(rf'^- \[[ xX]\] {n}\. ', l))
    i = next(j for j in range(start + 1, len(lines)) if re.match(r'^\s*\*\*Success Criteria:\*\*', lines[j]))
    out, fold = [], False
    for l in lines[i + 1:]:
        if CHECKBOX.match(l) or LABEL.match(l) or re.match(r'^- \[[ xX]\] \d+\. ', l): break
        if not l.strip(): fold = False; continue
        m = BULLET.match(l)
        if m: out.append(m.group(1)); fold = True
        elif fold and out: out[-1] += ' ' + l.strip()
    return out
def rows(doc):  # first cell of every row carrying a Status mark
    out = []
    for l in doc.split('\n'):
        if not l.startswith('|'): continue
        cells, cur, i = [], '', 0
        body = l.strip()[1:-1] if l.strip().endswith('|') else l.strip()[1:]
        while i < len(body):
            if body[i] == '\\' and i + 1 < len(body) and body[i + 1] == '|': cur += '\\|'; i += 2; continue
            if body[i] == '|': cells.append(cur); cur = ''
            else: cur += body[i]
            i += 1
        cells.append(cur)
        if len(cells) >= 3 and re.match(r'^\s*(✅|⚠️|❌)', cells[1]): out.append(cells[0])
    return out
tasks = show(f'{SPEC}/tasks.md'); tot_c = tot_r = ok = 0
for n in range(1, 19):
    c = collections.Counter(norm(x) for x in criteria(tasks, n))
    r = collections.Counter(norm(x) for x in rows(show(f'{SPEC}/completion/task-{n}-completion.md')))
    eq = c == r; ok += eq; tot_c += sum(c.values()); tot_r += sum(r.values())
    print(f'Task {n:>2}: criteria {sum(c.values()):>3} · rows {sum(r.values()):>3} · exact-set {"EQUAL" if eq else "DIFFER"}')
    if not eq:
        for x in (c - r): print('   only in tasks.md:', x[:160])
        for x in (r - c): print('   only in doc     :', x[:160])
print(f'TOTAL: {ok}/18 parents exact-set equal; {tot_c} criteria, {tot_r} rows')
```
